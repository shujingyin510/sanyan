// 类型标注测试

设 test = 导入("stdlib/test.san")

定义 加法 (x: 数字, y: 数字) {
    返回(x + y)
}

定义 问候 (名: 字符串) {
    返回(连接("你好，", 名))
}

// 正确的类型调用
设 加法结果 = 加法(1, 2)
输出(连接("加法: ", 加法结果))
test.断言相等(加法结果, 3)

设 问候结果 = 问候("世界")
输出(连接("问候: ", 问候结果))
test.断言相等(问候结果, "你好，世界")

// 错误的类型调用（应该被捕获）
尝试 {
    设 result = 加法("a", 1)
    输出(result)
    test.断言假(真)
} 捕获 (错误) {
    输出("类型检查通过：错误的类型被正确拒绝")
    test.断言真(真)
}

输出("类型标注测试完成")
